function(clean_protocol stem xml)
  foreach(mode client-header server-header private-code)
    if(mode STREQUAL "private-code")
      set(output "${CMAKE_CURRENT_BINARY_DIR}/${stem}-protocol.c")
    elseif(mode STREQUAL "client-header")
      set(output "${CMAKE_CURRENT_BINARY_DIR}/${stem}-client.h")
    else()
      set(output "${CMAKE_CURRENT_BINARY_DIR}/${stem}-server.h")
    endif()
    add_custom_command(OUTPUT "${output}" COMMAND "${WAYLAND_SCANNER}" "${mode}" "${xml}" "${output}" DEPENDS "${xml}" VERBATIM)
    list(APPEND generated "${output}")
  endforeach()
  set(PROTOCOL_FILES ${PROTOCOL_FILES} ${generated} PARENT_SCOPE)
endfunction()
